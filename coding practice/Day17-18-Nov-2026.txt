# Ansible to create multiple users 

- name: Create multiple users 
  hosts: all
  become: true

  vars:
    users:
      - dev1
      - dev2
      - dev3

  tasks:
    - name: create users
      user:
        name: "{{ item }}"
        state: present
     loop: "{{ users }}"

# jenkins simulated deployment

pipeline {
    agent any 
    parameter {
        choice(name: 'ENVIRONMENT', choices: ['dev', 'staging', 'prod'], description: 'Deployment Environment' )
    } 
    environment {
        INSTITUTE_NAME = "Pathnex" 
    }

    stages {
        stage('Deploy') {
            steps {
                sh 'echo "Deploying application "$INSTITUTE_NAME" to $ENVIRONMENT'
            }
        }
    }
}

Ansible - Installing apche and start service 

- name: Installing Apache and start service
  hosts: all
  become: yes
  tasks:
  - name: Installing Apache
    -yum: 
      name: httpd
      state: present

  - name: starting service 
    - service:
        name: httpd
        state: started
        enabled: yes

#kubernetes - pod creation with 3 replicas

apiVersion: apps/v1
kind: Deployment
metadata: 
  name: pathnex Deployment
spec:
  replicas: 3
  selector:
    matchLables:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app

    spec:
      containers:
        - name: app
          image: nginx
          ports:
           - containerPort: 80

#NodeJS build 
#jenkins
pipeline{
    agent any
    tools{
        nodejs "NodeJS-18"
    }
}

stages{
    stage('checkout'){
        steps{
            git url: 'https://github.com/Pathnex/sample-node-app.git' 
        }
    }

    stage('install dependencies'){
        steps{
            sh 'npm install'
        }
    }

    stage('Run Test'){
        steps{
            sh 'run test'
        }
    } 

    stage('Buidl){
        steps{
            sh 'npm run build'
        }
    }
}

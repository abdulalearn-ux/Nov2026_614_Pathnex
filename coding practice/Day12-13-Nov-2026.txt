# Ansible setup Auto scaling group

- name: setup Auto scaling group
  hosts: all
  become: yes

  tasks:
  - name: create launch config
    region: ap-south-1
    launch_config_name: pathnex-launch-config 
    min_size: 1
    max_size: 3
    desired_capacity: 2
    vpc_zone_identifier: subnet-987654321


# create a load balancer with ec2 instance

resource "aws_lb" "pathnex_lb" {
    name                = "pathnex-lb"
    internal            = false
    load_balancer_type  = "application"
    security_groups     = [aws_security_group.allow_ssh_http.id] 
    subnets             = [aws_subnet.public.id]
}

resource "aws_instance" "pathnex_ec2" {
    ami = "ami-01234abcdefghatni"
    instance_type = "t2.micro"
    security_groups = [aws_security_group.allow_ssh_http.id]

    tags = {
        Name = "Pathnex-EC2"
    }
}

# kubernetes Horizontal pod scaler

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-web-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-web-app 
  template:
    metadata:
      labels:
        app: pathnex-web-app 
    spec:
      containers:
        - name: web 
          image: nginx
          ports:
            - containerPorts: 80


apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa 
spec: 
  scaleTargetRef:
    apiVersion: apps/v1 
    kind: Deployment
    name: pathnex-web-app 
  minReplicas: 1
  maxReplicas: 5
  targetCPUUtilizationPercentage: 80

# jenkins implement parallel stages

    agent any

        stage('Build and Test') {

                stage('Build') {

            }

                stage('Test') {

                        echo "Test running" 
                    }
                }
            }

        stage('Deploy') {
            steps {
                echo 'Deploying project...'
            }    
           
        }
    }
  }
}

# GitLab CI parallel job build and Test

  - deploy

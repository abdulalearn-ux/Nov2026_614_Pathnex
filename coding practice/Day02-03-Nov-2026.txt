Day 02 - service and tags 

Ansible - Installing and Enabling Nginx

- name: Installing and Enabling Nginx
  hosts: all
  become: yes

  tasks:
  - name: Install Nginx
    yum:
      - name: nginx
        state: present
  
  - name: Enable Nginx
    service:
    - name: nginx
      state: started
      enabled: yes

# jenkins Pipeline - maven build

## Learn to compile, test, and package a java project using maven


Pipeline{

    agent any 
    tools{

        maven 'Maven-3.10.1'
        jdk 'JDK-25'
    }

    stages {

        stage('checkout'){
            steps{
                git url: 'https://github.com/Pathnex/java-app.git'
            }
        }

        stage('compile'){
            steps{
                sh 'mvn clean compile'
            }

        }

        stage('Test'){
            steps[
                sh 'mvn Test'
            ]
        }

        stage('package'){
            steps{
                sh 'mv package'
            }
        }
    }
}

# GitLab CI - Maven build

stages:
  - build
  - test
  - package


maven-build:
  stage: build
  image: maven:3.10.1-jdk-27
  script:
    - git clone https://github.com/Pathnex/java-app.git
    - cd java-app
    - mvn clean compile

maven-test:
  stage: test
  image: maven:3.10.1-jdk-27
  script:
  - cd java-app
  - mvn test

maven package:
  stage: package
  image: maven:3.10.1-jdk-27
  script:
    - cd java-app
    - mvn package

# ansible-  install + configure nginx
- name: install and configure nginx
  hosts: all
  become: yes

  tasks: 
  - name: install nginx
    yum:
      name: Nginx
      state: present
  - name: configure nginx
    template:
      src: nginx.conf.j2
      dest: /etc/nginx/nginx.conf
  - name: start nginx service
      service: 
        name: nginx
        state: started
        enabled: yes

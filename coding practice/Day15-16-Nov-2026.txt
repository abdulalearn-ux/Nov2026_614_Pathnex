    - name: start web app service
      service:
        - name: apache2
          state: started
          enabled: yes

# jenkins multi env deployment

pipeline {
    agent any
    environment {
        IMAGE_NAME = "pathnex-web-app"
        DEPLOY_ENV = "production"
    }

    stages {
        stage('Build') {
            steps {
                echo 'Building Docker Image...'
                docker.build("${IMAGE_NAME}")
            }
        }

        stage('Test') {
            steps {
                echo 'Testing'
            }
        }

        stage ('Deploy to kubernetes') {
            steps {
                script {
                    if (env.DEPLOY_ENV == 'production') {
                        sh 'kubectl apply -f kubernetes/prod-deployment.yaml'
                    }
                    else {
                        sh 'kubectl apply -f kubernetes/prod-dev.yaml'
                    }
                }
            }
        }
    }
}

# Ansible Install java and set JAVA_HOME

- name: install java on pathnex server
  hosts: all
  become: yes 

  tasks: 
  - name: Install java on pathnex server
    yum:
      name: java-11-openjdk
      state: present
  - name: set JAVA_HOME
    lineinfine:
      path: /etc/profile
      line: 'export JAVA_HOME=/usr/lib/jvm/java-11-openjdk'

# jenkins slack notification

pipeline {
    agent any 
    environment {
        INSTITUTE_NAME = "Pathnex"
        SLACK_CHANNEL = "#devops"

    }

    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }

        stage('Build') {
            steps {
                sh 'mv clean package'
            }
        }

    }

    post {
        success {
            slackSend(channel: "$SLACK_CHANNEL", message: "SUCCESS: $INSTITUTE_NAME Build #${env.BUILD_NUMBER}")
        }

        failure {
            slackSend(channel: "$SLACK_CHANNEL", message: "FAILURE: $INSTITUTE_NAME Build #${env.BUILD_NUMBER}")
        }
    }
     
}

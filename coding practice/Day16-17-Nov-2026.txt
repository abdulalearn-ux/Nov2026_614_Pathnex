# jenkins pipeline promoting artifacts

pipeline {
    agent any 
    environment {
        INSTITUTE_NAME = "Pathnex"
    }

    stages {
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }

        stage('promote')  {
            steps {
                input message: "Promote artifacts to staging?"
                sh 'echo "Promotinh artifacts for $INSTITUTE_NAME"'
            }
        }
    }
}
